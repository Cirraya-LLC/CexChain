edition 1
contract Notice
party shipper bound=creation{R3}
party consignee bound=creation{R3}
utc_offset +08:00{R25}
kind kind.thing Parcel
field field.holder instance party = shipper{R2}
field field.due instance date
field notified instance yes_no default val.no
field note instance text default ""
field tries instance number default 0
due field
state state.active first{R4}
state delivered{R5}
state done{R5}
state late{R5}
can deliver own by=shipper agents=delegation{R14}
  consent field.holder{R8}
  transition state.active -> delivered
  set field.holder consignee
  request notify to=consignee within=(duration time.day 1)
    set message "delivered"
    set tries tries
can close own by=consignee agents=delegation{R14}
  transition delivered -> done
when (and cond.overdue state.active{R7}) trigger=overdue function=when.remind{R24}
  request remind to=shipper within=(duration time.day 2)
answer notify done function=result
  set notified val.yes
  set note expr.result
answer notify fails function=result expired{R33}
  set note expr.reason
  set tries (+ tries 1)
  if (< tries 3)
    request notify to=shipper within=(duration time.day 2)
      set message "notice failed"
answer remind expires function=result
  transition -> late
test "notified on delivery"
  step shipper act.create creation expect=accepted
    set field.due 2027-03-10
  step shipper deliver at=2027-03-08 expect=state delivered
  step actor.executor act.notify answer=notify done expect=accepted
    set expr.result "MSG-1"
    expects notified val.yes
    expects note "MSG-1"
  step consignee close expect=state done
test "a failed notice, and an expired reminder"
  step shipper act.create creation expect=accepted
    set field.due 2027-03-10
  step actor.keeper remind rule=1 at=2027-03-11 expect=accepted
  step actor.keeper remind at=2027-03-13 answer=remind expires expect=state late
  step shipper deliver expect=accepted
  step actor.executor act.notify answer=notify fails expect=accepted
    set expr.reason "no_route"
    expects note "no_route"
    expects tries 1
  step actor.keeper act.notify at=2027-03-20 answer=notify expires expect=accepted
    expects tries 2
