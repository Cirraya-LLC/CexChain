edition 1
packs commerce
contract Agreement
party role.we bound=creation{R3}
party role.customer bound=creation{R3}
kind kind.thing Terms
field terms instance text
field price.price instance money cur.HKD
state state.proposed first{R4}
state signed{R5}
state closed{R5}
can sign own by=role.customer agents=delegation{R14}
  requires (cond.consent_of role.we)
  transition state.proposed -> signed
can close own by=role.we agents=delegation{R14}
  transition signed -> closed
test "signed and closed"
  step role.we act.create creation expect=accepted
    set terms "steel"
    set price.price (money cur.HKD 100)
  step role.customer sign expect=state signed
  step role.we close expect=state closed
test "the customer cannot close"
  step role.we act.create creation expect=accepted
    set terms "steel"
    set price.price (money cur.HKD 100)
  step role.customer close expect=refused "bo:state close"
  step role.customer sign expect=state signed
  step role.customer close expect=waits role.we
