edition 1
packs commerce
contract Batches
party role.we bound=creation{R3}
party role.customer bound=creation{R3}
kind kind.thing Batch
field codes instance list 3 text
field sizes instance list 16 number{R16}
field total instance number
field signers instance list 4 text
state state.active first{R4}
can load own by=role.we agents=delegation{R14}
  input rows instance list 4 number
  input tags instance list 16 text{R16}
  requires (> (expr.count_of rows) 0)
  requires (< (expr.sum_of rows) 100)
  each n in=rows
    requires (> n 0)
    set total (+ total n)
  each t in=tags
    if (in t (list "x" "y"))
      reject "no x or y"
  requires (not (in "bad" tags))
  set sizes rows
  set codes (list "a" "b")
can mark own by=role.we agents=delegation{R14}
  input code instance text
  requires (in code codes)
  set signers (list "z"){R29}
  each s in=(list 1 2 3)
    requires (< s 5)
