packs commerce finance
contract Loan
party role.we bound=creation{R3}
party role.customer bound=creation{R3}
party role.bank bound=lend{R3}
kind kind.thing Note
field field.amount instance money cur.HKD
field field.holder instance party = role.customer{R2}
state state.proposed first{R4}
state lent{R5}
state repaid{R5}
can lend own by=role.we to=role.bank agents=delegation{R14}
  consent role.bank{R3}
  requires (cond.consent_of role.customer)
  transition state.proposed -> lent
can repay own by=role.bank agents=delegation{R14}
  requires (= field.holder role.customer)
  transition lent -> repaid
test "lent and repaid"
  step role.we act.create creation expect=accepted
    set field.amount (money cur.HKD 1000)
  step role.we lend to=ledger:bank1 at=2027-01-06 in_person expect=state lent
  step ledger:bank1 repay agent expect=state repaid
test "unhappy"
  step role.we act.create creation at=2027-01-06 expect=accepted
    set field.amount (money cur.HKD 1000)
  step role.we lend to=ledger:bank1 expect=waits role.customer ledger:bank1
  step ledger:bank1 repay expect=refused "bo:party bank"
  step role.customer lend to=ledger:bank1 expect=waits role.we
  step role.we lend to=ledger:bank1 expect=state lent
  step role.we lend to=ledger:bank1 expect=refused
